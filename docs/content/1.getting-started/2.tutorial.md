    ### Create a PocketBase instance

    ```ts [example.ts]
    import PocketBase from 'pocketbase';

    const pb = new PocketBase("https://example.com");
    ```

    ### Define your types

    ```ts [types.ts]
    export interface User {
      id: string;
      name: string;
      email: string;
      created: Date;
      updated: Date;
    }

    export interface Post {
      id: string;
      title: string;
      content: string;
      author: User;
      status: 'draft' | 'published';
      created: Date;
      updated: Date;
    }
    ```

    You can read more about how to set up you types in [Setting up Types](/getting-started/setting-up-types).

    ### Build a type-safe query

    ```ts [example.ts]
    import { pbQuery } from '@sergio9929/pb-query';
    import type { Post } from './types';

    const query = pbQuery<Post>() // posts collection
      .fields([
        'title',
        'content:excerpt(100,true)',
        'status',
        'author',
        'expand.author.name', // Automatically expanded
      ])
      .search(['title', 'content', 'author.name'], 'footba')
      .and()
      .group((q) =>
        q.equal('status', 'published')
          .or()
          .equal('author.name', 'Sergio')
      )
      .sort(['-created'])
      .build(pb.filter);
    ```

    Now your `query` object contains the properly formatted [`filter`](/returning-values/filter), [`expand`](/returning-values/expand), [`fields`](/returning-values/fields), and [`sort`](/returning-values/sort) strings.

    You can read more about the output returned by `pbQuery()` in [Returning Values](/returning-values/fields).

    ### Use your query

    ```ts [example.ts]
    const records = await pb.collection("posts").getList(1, 20, query);
    ```

## Full example

::code-tree{default-value="app/example.ts"}

```ts [app/example.ts]
import PocketBase from 'pocketbase';
import { pbQuery } from '@sergio9929/pb-query';
import type { Post } from './types';

const pb = new PocketBase("https://example.com");

const query = pbQuery<Post>() // posts collection
  .fields([
    'title',
    'content:excerpt(100,true)',
    'status',
    'author',
    'expand.author.name', // Automatically expanded
  ])
  .search(['title', 'content', 'author.name'], 'footba')
  .and()
  .group((q) =>
    q.equal('status', 'published')
      .or()
      .equal('author.name', 'Sergio')
  )
  .sort(['-created'])
  .build(pb.filter);

const records = await pb.collection("posts").getList(1, 20, query);
```

```ts [app/types.ts]
export interface User {
  id: string;
  name: string;
  email: string;
  created: Date;
  updated: Date;
}

export interface Post {
  id: string;
  title: string;
  content: string;
  author: User;
  status: 'draft' | 'published';
  created: Date;
  updated: Date;
}
```

```js [pocketbase/pb_hooks/example.pb.js]
/// <reference path="../pb_data/types.d.ts" />

routerAdd("GET", "/example", (e) => {
  const { pbQuery } = require('@sergio9929/pb-query');

  const { filter, sort } = pbQuery()
    .fields([
      'title',
      'content:excerpt(100,true)',
      'status',
      'author',
      'expand.author.name', // Automatically expanded
    ])
    .search(['title', 'content', 'author.name'], 'footba')
    .and()
    .group((q) =>
      q.equal('status', 'published')
        .or()
        .equal('author.name', 'Sergio')
    )
    .sort(['-created'])
    .build(pb.filter);

  const records = $app.findRecordsByFilter(
    'posts',
    filter.raw,
    sort,
    20,
    0,
    filter.values,
  );

  return e.json(200, records);
});
```
